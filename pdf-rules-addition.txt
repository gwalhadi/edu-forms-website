    // Original PDFs of "PDF forms" (blank templates uploaded by the owner, never teachers' answers).
    // Anyone may read a file by its random id (teachers need it to fill the form); only the owner writes.
    match /pdfFiles/{fileId} {
      allow get: if true;
      allow list: if isAdmin();
      allow create, update: if isAdmin()
        && fileId.matches('^[A-Za-z0-9_-]{8,40}$')
        && request.resource.data.keys().hasOnly(['name', 'size', 'chunks', 'createdAt'])
        && request.resource.data.name is string && request.resource.data.name.size() <= 200
        && request.resource.data['size'] is int && request.resource.data['size'] <= 10485760
        && request.resource.data['chunks'] is int && request.resource.data['chunks'] >= 1 && request.resource.data['chunks'] <= 20;
      allow delete: if isAdmin();

      match /chunks/{n} {
        allow get: if true;
        allow list: if isAdmin();
        allow create, update: if isAdmin()
          && n.matches('^[0-9]{1,2}$')
          && request.resource.data.keys().hasOnly(['data'])
          && request.resource.data.data is string
          && request.resource.data.data.size() <= 900000;
        allow delete: if isAdmin();
      }
    }
